package com.example.chessoverlay

import kotlin.math.abs
import kotlin.math.max

data class Move(val from: Int, val to: Int, val promo: Char = ' ') {
    override fun toString(): String {
        val s = "" + ('a' + from % 8) + (8 - from / 8) + ('a' + to % 8) + (8 - to / 8)
        return if (promo != ' ') s + promo else s
    }
}

class Position(
    val b: CharArray = CharArray(64) { '.' },
    var white: Boolean = true,
    var castle: String = "KQkq",
    var ep: Int = -1
) {
    companion object {
        private val DIRS = arrayOf(
            intArrayOf(-1, 0), intArrayOf(1, 0), intArrayOf(0, -1), intArrayOf(0, 1),
            intArrayOf(-1, -1), intArrayOf(-1, 1), intArrayOf(1, -1), intArrayOf(1, 1)
        )
        private val KN = arrayOf(
            intArrayOf(-2, -1), intArrayOf(-2, 1), intArrayOf(-1, -2), intArrayOf(-1, 2),
            intArrayOf(1, -2), intArrayOf(1, 2), intArrayOf(2, -1), intArrayOf(2, 1)
        )
        const val START = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"

        fun fromFen(fen: String): Position? {
            try {
                val p = fen.trim().split(Regex("\\s+"))
                val rows = p[0].split("/")
                if (rows.size != 8) return null
                val pos = Position()
                for (r in 0 until 8) {
                    var f = 0
                    for (c in rows[r]) {
                        if (c.isDigit()) f += c - '0'
                        else {
                            if (f > 7 || "pnbrqkPNBRQK".indexOf(c) < 0) return null
                            pos.b[r * 8 + f] = c
                            f++
                        }
                    }
                    if (f != 8) return null
                }
                pos.white = p.getOrElse(1) { "w" } == "w"
                val cs = p.getOrElse(2) { "-" }
                pos.castle = if (cs == "-") "" else cs
                val e = p.getOrElse(3) { "-" }
                pos.ep = if (e.length == 2) (8 - (e[1] - '0')) * 8 + (e[0] - 'a') else -1
                if (pos.b.indexOf('K') < 0 || pos.b.indexOf('k') < 0) return null
                return pos
            } catch (ex: Exception) {
                return null
            }
        }
    }

    private fun addPawn(res: MutableList<Move>, from: Int, to: Int, promo: Boolean) {
        if (promo) {
            for (p in "qrbn") res.add(Move(from, to, p))
        } else res.add(Move(from, to))
    }

    fun pseudo(side: Boolean): MutableList<Move> {
        val res = ArrayList<Move>()
        for (sq in 0 until 64) {
            val c = b[sq]
            if (c == '.' || c.isUpperCase() != side) continue
            val r = sq / 8
            val f = sq % 8
            when (c.lowercaseChar()) {
                'p' -> {
                    // белые идут вверх (row уменьшается), чёрные вниз (row растёт)
                    val dir = if (side) -1 else 1
                    val startRow = if (side) 6 else 1
                    val lastRow = if (side) 0 else 7
                    val r1 = r + dir
                    if (r1 in 0..7) {
                        // прямо — только на пустую клетку
                        if (b[r1 * 8 + f] == '.') {
                            addPawn(res, sq, r1 * 8 + f, r1 == lastRow)
                            val r2 = r + 2 * dir
                            if (r == startRow && b[r2 * 8 + f] == '.') res.add(Move(sq, r2 * 8 + f))
                        }
                        // по диагонали — только взятие (или en passant)
                        for (df in intArrayOf(-1, 1)) {
                            val nf = f + df
                            if (nf < 0 || nf > 7) continue
                            val t = r1 * 8 + nf
                            val tc = b[t]
                            if (tc != '.' && tc.isUpperCase() != side) {
                                addPawn(res, sq, t, r1 == lastRow)
                            } else if (tc == '.' && t == ep) {
                                val victim = b[r * 8 + nf]
                                if (victim.lowercaseChar() == 'p' && victim.isUpperCase() != side) res.add(Move(sq, t))
                            }
                        }
                    }
                }
                'n' -> for (d in KN) {
                    val nr = r + d[0]
                    val nf = f + d[1]
                    if (nr < 0 || nr > 7 || nf < 0 || nf > 7) continue
                    val tc = b[nr * 8 + nf]
                    if (tc == '.' || tc.isUpperCase() != side) res.add(Move(sq, nr * 8 + nf))
                }
                'k' -> {
                    for (d in DIRS) {
                        val nr = r + d[0]
                        val nf = f + d[1]
                        if (nr < 0 || nr > 7 || nf < 0 || nf > 7) continue
                        val tc = b[nr * 8 + nf]
                        if (tc == '.' || tc.isUpperCase() != side) res.add(Move(sq, nr * 8 + nf))
                    }
                    val base = if (side) 56 else 0
                    val ks = if (side) "K" else "k"
                    val qs = if (side) "Q" else "q"
                    if (sq == base + 4 && attacked(sq, side.not()).not()) {
                        if (castle.contains(ks) && b[base + 7] == (if (side) 'R' else 'r') &&
                            b[base + 5] == '.' && b[base + 6] == '.' &&
                            attacked(base + 5, side.not()).not() && attacked(base + 6, side.not()).not()
                        ) res.add(Move(sq, base + 6))
                        if (castle.contains(qs) && b[base] == (if (side) 'R' else 'r') &&
                            b[base + 1] == '.' && b[base + 2] == '.' && b[base + 3] == '.' &&
                            attacked(base + 3, side.not()).not() && attacked(base + 2, side.not()).not()
                        ) res.add(Move(sq, base + 2))
                    }
                }
                else -> {
                    val lc = c.lowercaseChar()
                    for (i in 0 until 8) {
                        if (lc == 'b' && i < 4) continue
                        if (lc == 'r' && i >= 4) continue
                        var nr = r + DIRS[i][0]
                        var nf = f + DIRS[i][1]
                        while (nr in 0..7 && nf in 0..7) {
                            val tc = b[nr * 8 + nf]
                            if (tc == '.') res.add(Move(sq, nr * 8 + nf))
                            else {
                                if (tc.isUpperCase() != side) res.add(Move(sq, nr * 8 + nf))
                                break
                            }
                            nr += DIRS[i][0]
                            nf += DIRS[i][1]
                        }
                    }
                }
            }
        }
        return res
    }

    fun attacked(sq: Int, by: Boolean): Boolean {
        val r = sq / 8
        val f = sq % 8
        val pr = if (by) r + 1 else r - 1
        if (pr in 0..7) {
            for (df in intArrayOf(-1, 1)) {
                val nf = f + df
                if (nf in 0..7 && b[pr * 8 + nf] == (if (by) 'P' else 'p')) return true
            }
        }
        for (d in KN) {
            val nr = r + d[0]
            val nf = f + d[1]
            if (nr in 0..7 && nf in 0..7 && b[nr * 8 + nf] == (if (by) 'N' else 'n')) return true
        }
        for (d in DIRS) {
            val nr = r + d[0]
            val nf = f + d[1]
            if (nr in 0..7 && nf in 0..7 && b[nr * 8 + nf] == (if (by) 'K' else 'k')) return true
        }
        for (i in 0 until 8) {
            var nr = r + DIRS[i][0]
            var nf = f + DIRS[i][1]
            while (nr in 0..7 && nf in 0..7) {
                val tc = b[nr * 8 + nf]
                if (tc != '.') {
                    if (tc.isUpperCase() == by) {
                        val l = tc.lowercaseChar()
                        if (l == 'q' || (l == 'r' && i < 4) || (l == 'b' && i >= 4)) return true
                    }
                    break
                }
                nr += DIRS[i][0]
                nf += DIRS[i][1]
            }
        }
        return false
    }

    fun inCheck(side: Boolean): Boolean {
        val k = b.indexOf(if (side) 'K' else 'k')
        return k >= 0 && attacked(k, side.not())
    }

    fun make(m: Move): Position {
        val n = Position(b.copyOf(), white.not(), castle, -1)
        val c = n.b[m.from]
        val w = c.isUpperCase()
        val tc = n.b[m.to]
        n.b[m.to] = c
        n.b[m.from] = '.'
        if (m.promo != ' ') n.b[m.to] = if (w) m.promo.uppercaseChar() else m.promo.lowercaseChar()
        val fr = m.from / 8
        val ff = m.from % 8
        val tr = m.to / 8
        val tf = m.to % 8
        if (c.lowercaseChar() == 'p') {
            if (m.to == ep && ff != tf && tc == '.') n.b[fr * 8 + tf] = '.'
            if (abs(tr - fr) == 2) n.ep = ((fr + tr) / 2) * 8 + ff
        }
        if (c.lowercaseChar() == 'k' && abs(tf - ff) == 2) {
            if (tf == 6) {
                n.b[tr * 8 + 5] = n.b[tr * 8 + 7]
                n.b[tr * 8 + 7] = '.'
            } else {
                n.b[tr * 8 + 3] = n.b[tr * 8]
                n.b[tr * 8] = '.'
            }
        }
        var cs = castle
        if (c == 'K') cs = cs.replace("K", "").replace("Q", "")
        if (c == 'k') cs = cs.replace("k", "").replace("q", "")
        for (s in intArrayOf(m.from, m.to)) {
            when (s) {
                63 -> cs = cs.replace("K", "")
                56 -> cs = cs.replace("Q", "")
                7 -> cs = cs.replace("k", "")
                0 -> cs = cs.replace("q", "")
            }
        }
        n.castle = cs
        return n
    }

    fun legal(): List<Move> = pseudo(white).filter { m ->
        val n = make(m)
        n.inCheck(white).not()
    }
}

object Engine {
    private class TimeUp : RuntimeException()

    private var deadline = 0L
    private var nodes = 0

    private fun tick() {
        nodes++
        if ((nodes and 1023) == 0 && System.currentTimeMillis() > deadline) throw TimeUp()
    }

    private fun value(c: Char): Int = when (c.lowercaseChar()) {
        'p' -> 100; 'n' -> 320; 'b' -> 330; 'r' -> 500; 'q' -> 900; else -> 0
    }

    private fun eval(p: Position): Int {
        var s = 0
        for (sq in 0 until 64) {
            val c = p.b[sq]
            if (c == '.') continue
            val r = sq / 8
            val f = sq % 8
            var v = value(c)
            val l = c.lowercaseChar()
            if (l == 'p' || l == 'n' || l == 'b') {
                val dist = abs(r - 3.5) + abs(f - 3.5)
                v += (10 - dist * 3).toInt()
            }
            if (l == 'p') v += if (c.isUpperCase()) (6 - r) * 4 else (r - 1) * 4
            s += if (c.isUpperCase()) v else -v
        }
        return if (p.white) s else -s
    }

    private fun order(p: Position, moves: List<Move>): List<Move> =
        moves.sortedByDescending { m ->
            val t = p.b[m.to]
            (if (t != '.') value(t) * 10 - value(p.b[m.from]) else 0) + (if (m.promo == 'q') 800 else 0)
        }

    private fun qsearch(p: Position, alpha0: Int, beta: Int, qd: Int): Int {
        tick()
        val stand = eval(p)
        if (qd <= 0) return stand
        if (stand >= beta) return beta
        var alpha = max(alpha0, stand)
        val caps = p.pseudo(p.white).filter { p.b[it.to] != '.' || it.promo == 'q' }
        for (m in order(p, caps)) {
            val n = p.make(m)
            if (n.inCheck(p.white)) continue
            val s = -qsearch(n, -beta, -alpha, qd - 1)
            if (s >= beta) return beta
            if (s > alpha) alpha = s
        }
        return alpha
    }

    private fun search(p: Position, d: Int, alpha0: Int, beta: Int, ply: Int): Int {
        tick()
        if (d <= 0) return qsearch(p, alpha0, beta, 3)
        val ms = p.legal()
        if (ms.isEmpty()) return if (p.inCheck(p.white)) -100000 + ply else 0
        var alpha = alpha0
        for (m in order(p, ms)) {
            val s = -search(p.make(m), d - 1, -beta, -alpha, ply + 1)
            if (s >= beta) return beta
            if (s > alpha) alpha = s
        }
        return alpha
    }

    /** Итеративное углубление до depth; при нехватке времени возвращает лучший ход предыдущей глубины. */
    fun best(pos: Position, depth: Int, timeMs: Long = 8000L): Move? {
        deadline = System.currentTimeMillis() + timeMs
        nodes = 0
        var ms = order(pos, pos.legal())
        if (ms.isEmpty()) return null
        var best: Move = ms[0]
        for (d in 1..depth) {
            try {
                var alpha = -1000000
                var cur: Move? = null
                for (m in ms) {
                    val s = -search(pos.make(m), d - 1, -1000000, -alpha, 1)
                    if (cur == null || s > alpha) {
                        alpha = s
                        cur = m
                    }
                }
                if (cur != null) {
                    best = cur
                    val c = cur
                    ms = listOf(c) + ms.filter { it != c }
                }
            } catch (e: TimeUp) {
                break
            }
        }
        return best
    }
}

    private fun makeDraggable(handle: View, lp: WindowManager.LayoutParams, target: View, onClick: (() -> Unit)?) {
        var dx = 0f
        var dy = 0f
        var sx = 0
        var sy = 0
        var moved = false
        handle.setOnTouchListener { _, e ->
            when (e.action) {
                MotionEvent.ACTION_DOWN -> {
                    dx = e.rawX; dy = e.rawY; sx = lp.x; sy = lp.y; moved = false
                }
                MotionEvent.ACTION_MOVE -> {
                    val mx = (e.rawX - dx).toInt()
                    val my = (e.rawY - dy).toInt()
                    if (Math.abs(mx) > dp(4) || Math.abs(my) > dp(4)) moved = true
                    if (moved) {
                        lp.x = sx + mx
                        lp.y = sy + my
                        wm.updateViewLayout(target, lp)
                    }
                }
                MotionEvent.ACTION_UP -> if (moved.not()) onClick?.invoke()
            }
            true
        }
    }

    /** Широкая плашка двигается только вверх/вниз. */

        handle.setOnTouchListener { _, e ->

                MotionEvent.ACTION_DOWN -> {
                    dy = e.rawY; sy = lp.y
                }
                MotionEvent.ACTION_MOVE -> {

                    lp.y = (sy + (e.rawY - dy).toInt()).coerceIn(0, maxY)
                    try {
                        wm.updateViewLayout(target, lp)
                    } catch (ex: Exception) {
                    }
                }
            }
            true
        }
    }

    private fun captureBoard(fullHide: Boolean, done: (Bitmap?, String) -> Unit) {

            return

            hid = true

                menu.visibility = View.INVISIBLE

    private fun grab(b: RectF, hideAt: Long, attempt: Int, needFresh: Boolean, done: (Bitmap?, String) -> Unit) {
        val cur = lastImg
        val fresh = cur != null && (needFresh.not() || lastFrameAt >= hideAt + 80)
        if (fresh.not() && attempt < 10) {
            handler.postDelayed({ grab(b, hideAt, attempt + 1, needFresh, done) }, 150L)
            return
        }
        if (cur == null) {
            setUiVisible(true)
            done(null, "Нет кадра экрана.")
            return
        }
        var bmp: Bitmap? = null
        try {
            val plane = cur.planes[0]
            val bw = plane.rowStride / plane.pixelStride
            val full = Bitmap.createBitmap(bw, cur.height, Bitmap.Config.ARGB_8888)
            val buf = plane.buffer
            buf.rewind()
            full.copyPixelsFromBuffer(buf)
            val l = b.left.toInt().coerceIn(0, cur.width - 2)
            val t = b.top.toInt().coerceIn(0, cur.height - 2)
            val w = b.width().toInt().coerceAtMost(cur.width - l).coerceAtLeast(1)
            val h = b.height().toInt().coerceAtMost(cur.height - t).coerceAtLeast(1)
            bmp = Bitmap.createBitmap(full, l, t, w, h)
            full.recycle()
        } catch (e: Exception) {
            bmp = null
        }
        // сам кадр не закрываем: он остаётся «последним» и закроется, когда придёт новый (только RAM)
        setUiVisible(true)
        val out = bmp
        if (out == null) done(null, "Ошибка скриншота.") else done(out, "")
    }

    private fun onRecognized(rec: Rec, auto: Boolean, white: Boolean) {
        val fen = rec.fen
        val pos = if (fen != null) Position.fromFen(fen) else null
        if (fen == null || pos == null) {
            status.text = (if (auto) "Авто: " else "") + (if (fen == null) rec.msg else "Некорректный FEN: $fen")
            busy = false
            return
        }
        val placement = fen.substringBefore(' ')
        if (auto) {
            // мы уже сходили, соперник ещё нет
            if (placement == afterOurMove) {
                lastPlacement = placement
                status.text = "Авто: жду ход соперника"
                busy = false
                return
            }
            // ход не прошёл (позиция та же, что до нашего хода) — разрешаем повтор
            val retry = autoPlay && playedFrom != null && placement == playedFrom && retries < 2 &&
                    SystemClock.uptimeMillis() - lastMoveAt > 5000L
            if (placement == lastPlacement && retry.not()) {
                busy = false
                return
            }
            val ourTurn = if (placement == BoardRecognizer.START_PLACEMENT) white else rec.turn != 0
            lastPlacement = placement
            if (ourTurn.not()) {
                status.text = "Авто: ход соперника"
                arrow.move = null
                arrow.invalidate()
                busy = false
                return
            }
        }
        lastPlacement = placement
        pos.white = white // ищем лучший ход для выбранной стороны
        runEngine(pos, rec.msg + " ", placement, auto)
    }

    private fun runEngine(pos: Position, prefix: String, placement: String, auto: Boolean) {
        status.text = prefix + "Думаю..."
        val d = depths[level]
        Thread {
            val mv = Engine.best(pos, d)
            handler.post { onBest(pos, mv, prefix, placement, auto) }
        }.start()
    }

    private fun onBest(pos: Position, mv: Move?, prefix: String, placement: String, auto: Boolean) {
        if (mv == null) {
            status.text = prefix + "Нет ходов"
            busy = false
            return
        }
        if (autoPlay && auto) {
            // стрелку не рисуем; ждём, чтобы соперник/анимация закончились, и перепроверяем позицию
            arrow.move = null
            arrow.invalidate()
            status.text = "Автоход: нашёл ${fmt(mv)}, жду 2.5 с..."
            handler.postDelayed({ confirmAndPlay(pos, mv, placement) }, 2500L)
        } else {
            arrow.move = mv
            arrow.flip = playWhite.not()
            arrow.invalidate()
            status.text = prefix + "Лучший ход: ${fmt(mv)}"
            busy = false
        }
    }

    private fun confirmAndPlay(pos: Position, mv: Move, placement: String) {
        if (autoPlay.not()) {
            busy = false
            return
        }
        captureBoard(false) { bmp, msg ->
            if (bmp == null) {
                busy = false
                status.text = "Автоход: $msg"
            } else {
                val white = playWhite
                Thread {
                    val rec = BoardRecognizer.recognize(bmp, white, prefs)
                    bmp.recycle()
                    handler.post {
                        val p2 = rec.fen?.substringBefore(' ')
                        if (p2 == placement) {
                            executeMove(pos, mv, placement)
                        } else {
                            status.text = "Автоход: позиция изменилась, пересчитаю"
                            lastPlacement = null
                            busy = false
                        }
                    }
                }.start()
            }
        }
    }

    private fun sqCenter(sq: Int): FloatArray {
        val f = sq % 8
        val r = sq / 8
        val b = arrow.board
        val fl = playWhite.not()
        val x = b.left + ((if (fl) 7 - f else f) + 0.5f) * b.width() / 8f
        val y = b.top + ((if (fl) 7 - r else r) + 0.5f) * b.height() / 8f
        return floatArrayOf(x, y)
    }

    private fun executeMove(pos: Position, mv: Move, placementBefore: String) {
        val ap = AutoPlayer.instance
        if (ap == null) {
            autoPlay = false
            restartTick()
            updateToggles()

            return

        val b = sqCenter(mv.to)
        setTouchable(false)
        status.text = "Автоход: делаю ${fmt(mv)}..."
        ap.playMove(a[0], a[1], b[0], b[1], mv.promo != ' ') { ok ->

                lastMoveAt = SystemClock.uptimeMillis()

                    playedFrom = placementBefore

                } else {
                    status.text = "Автоход: жест не выполнен"
                }
                busy = false
            }
        }
    }

    fun loadTemplates(p: SharedPreferences) {

    /** Расстановка фигур (первое поле FEN) из массива 8x8, индекс = (8 - ранг) * 8 + файл. */
    fun placement(b: CharArray): String {
        val sb = StringBuilder()
        for (rr in 0..7) {
            var empty = 0
            for (ff in 0..7) {
                val ch = b[rr * 8 + ff]
                if (ch == '.') empty++
                else {
                    if (empty > 0) {
                        sb.append(empty)
                        empty = 0
                    }
                    sb.append(ch)
                }
            }
            if (empty > 0) sb.append(empty)
            if (rr < 7) sb.append('/')
        }
        return sb.toString()
    }

    /**
     * Клетка -> (занята?, силуэт 32x32, яркость внутри силуэта, цвет фона клетки).
     * Маска = цветовое отличие от фона ИЛИ морфологический градиент яркости, пороги адаптивные
     * (считаются по шуму рамки клетки), поэтому тёмные темы тоже работают.
     */
    private fun analyze(cell: Mat): Cell {
        val gray = Mat()
        Imgproc.cvtColor(cell, gray, Imgproc.COLOR_BGR2GRAY)

        val buf = ByteArray(S * S * 3)
        cell.get(0, 0, buf)
        val bs = ArrayList<Int>()
        val gs = ArrayList<Int>()
        val rs = ArrayList<Int>()
        forRing { x, y ->
            val i = (y * S + x) * 3
            bs.add(buf[i].toInt() and 0xFF)
            gs.add(buf[i + 1].toInt() and 0xFF)
            rs.add(buf[i + 2].toInt() and 0xFF)
        }
        val medB = med(bs)
        val medG = med(gs)
        val medR = med(rs)
        val bgI = intArrayOf(medB, medG, medR)
        val bg = Scalar(medB.toDouble(), medG.toDouble(), medR.toDouble())
        val bgMat = Mat(S, S, CvType.CV_8UC3, bg)
        val diff = Mat()
        Core.absdiff(cell, bgMat, diff)
        val ch = ArrayList<Mat>()
        Core.split(diff, ch)
        val dmax = Mat()
        Core.max(ch[0], ch[1], dmax)
        Core.max(dmax, ch[2], dmax)

        val blur = Mat()
        Imgproc.GaussianBlur(gray, blur, Size(3.0, 3.0), 0.0)
        val k3 = Imgproc.getStructuringElement(Imgproc.MORPH_ELLIPSE, Size(3.0, 3.0))
        val k5 = Imgproc.getStructuringElement(Imgproc.MORPH_ELLIPSE, Size(5.0, 5.0))
        val grad = Mat()
        Imgproc.morphologyEx(blur, grad, Imgproc.MORPH_GRADIENT, k3)

        val db = ByteArray(S * S)
        val gb = ByteArray(S * S)
        dmax.get(0, 0, db)
        grad.get(0, 0, gb)
        val dRing = ArrayList<Int>()
        val gRing = ArrayList<Int>()
        forRing { x, y ->
            val i = y * S + x
            dRing.add(db[i].toInt() and 0xFF)
            gRing.add(gb[i].toInt() and 0xFF)
        }
        val td = (p90(dRing) * 2 + 6).coerceIn(14, 45)
        val tg = (p90(gRing) * 2 + 8).coerceIn(10, 40)
        val md = Mat()
        val mg = Mat()
        Imgproc.threshold(dmax, md, td.toDouble(), 255.0, Imgproc.THRESH_BINARY)
        Imgproc.threshold(grad, mg, tg.toDouble(), 255.0, Imgproc.THRESH_BINARY)
        val mask = Mat()
        Core.bitwise_or(md, mg, mask)

        val roi = Mat.zeros(S, S, CvType.CV_8UC1)
        roi.submat(Rect(4, 4, S - 8, S - 8)).setTo(Scalar(255.0))
        Core.bitwise_and(mask, roi, mask)

        Imgproc.morphologyEx(mask, mask, Imgproc.MORPH_CLOSE, k5)
        val contours = ArrayList<MatOfPoint>()
        Imgproc.findContours(mask.clone(), contours, Mat(), Imgproc.RETR_EXTERNAL, Imgproc.CHAIN_APPROX_SIMPLE)
        var maxA = 0.0
        var big: MatOfPoint? = null
        for (c in contours) {
            val a = Imgproc.contourArea(c)
            if (a > maxA) {
                maxA = a
                big = c
            }
        }
        val bigC = big ?: return Cell(false, null, 0f, bgI)
        val br = Imgproc.boundingRect(bigC)
        if (maxA < 0.08 * S * S || br.height < 0.38 * S) return Cell(false, null, 0f, bgI)

        val keep = ArrayList<MatOfPoint>()
        for (c in contours) {
            val a = Imgproc.contourArea(c)
            if (a >= maxOf(60.0, 0.12 * maxA)) keep.add(c)
        }
        val filled = Mat.zeros(S, S, CvType.CV_8UC1)
        Imgproc.drawContours(filled, keep, -1, Scalar(255.0), -1)

        val inner = Mat()
        Imgproc.erode(filled, inner, k5)
        Imgproc.erode(inner, inner, k5)
        val src = if (Core.countNonZero(inner) < 30) filled else inner
        val bright = Core.mean(gray, src).`val`[0]

        val small = Mat()
        Imgproc.resize(filled, small, Size(N.toDouble(), N.toDouble()), 0.0, 0.0, Imgproc.INTER_AREA)
        val sb = ByteArray(N * N)
        small.get(0, 0, sb)
        val m = FloatArray(N * N) { (sb[it].toInt() and 0xFF) / 255f }
        return Cell(true, m, bright.toFloat(), bgI)
    }

    private fun softIou(a: FloatArray, b: FloatArray): Float {
        var mn = 0f
        var mx = 0f
        for (i in a.indices) {
            mn += minOf(a[i], b[i])
            mx += maxOf(a[i], b[i])
        }
        return if (mx <= 0f) 0f else mn / mx
    }

    /** Скриншот доски (только в памяти) -> сетка 8x8 проанализированных клеток. */
    private fun scanCells(bmp: Bitmap): Array<Array<Cell>> {
        val rgba = Mat()
        val bgr = Mat()
        val board = Mat()
        try {
            Utils.bitmapToMat(bmp, rgba)
            Imgproc.cvtColor(rgba, bgr, Imgproc.COLOR_RGBA2BGR)
            Imgproc.resize(bgr, board, Size((S * 8).toDouble(), (S * 8).toDouble()), 0.0, 0.0, Imgproc.INTER_AREA)
            return Array(8) { r -> Array(8) { c -> analyze(board.submat(Rect(c * S, r * S, S, S)).clone()) } }
        } finally {
            rgba.release()
            bgr.release()
            board.release()
        }
    }

    // ---------- автообучение эталонов по начальной позиции ----------

    private fun startMismatch(cells: Array<Array<Cell>>, flip: Boolean): Pair<ArrayList<String>, ArrayList<String>> {
        val startB = Position.fromFen(Position.START)?.b ?: CharArray(64) { '.' }
        val missing = ArrayList<String>()
        val extra = ArrayList<String>()
        for (r in 0..7) {
            for (c in 0..7) {
                val rr = if (flip) 7 - r else r
                val ff = if (flip) 7 - c else c
                val exp = startB[rr * 8 + ff]
                val occ = cells[r][c].occupied
                if (exp != '.' && occ.not()) missing.add(sqName(rr, ff))
                if (exp == '.' && occ) extra.add(sqName(rr, ff))
            }
        }
        return Pair(missing, extra)
    }

    private fun learnFromCells(cells: Array<Array<Cell>>, flip: Boolean, prefs: SharedPreferences): LearnRes {
        val startB = Position.fromFen(Position.START)?.b ?: return LearnRes(false, "Внутренняя ошибка START")
        val mm = startMismatch(cells, flip)
        if (mm.first.size + mm.second.size > 4) {
            return LearnRes(false, "Не похоже на начальную позицию. Проверь рамку («Доска») и переключатель «Играю за».")
        }
        val sums = HashMap<Char, FloatArray>()
        val cnt = HashMap<Char, Int>()
        var sw = 0f
        var nw = 0
        var sbk = 0f
        var nb = 0
        var samples = 0
        for (r in 0..7) {
            for (c in 0..7) {
                val cell = cells[r][c]
                val m = cell.mask ?: continue
                val rr = if (flip) 7 - r else r
                val ff = if (flip) 7 - c else c
                val ch = startB[rr * 8 + ff]
                if (ch == '.') continue
                val t = ch.lowercaseChar()
                val acc = sums.getOrPut(t) { FloatArray(N * N) }
                for (i in 0 until N * N) acc[i] += m[i]
                cnt[t] = (cnt[t] ?: 0) + 1
                samples++
                if (ch.isUpperCase()) {
                    sw += cell.bright; nw++
                } else {
                    sbk += cell.bright; nb++
                }
            }
        }
        if (sums.size < 6) {
            val lack = "pnbrqk".filter { sums.containsKey(it).not() }
            return LearnRes(false, "Не хватило фигур для эталонов (нет типов: $lack). Нет фигур на: " + mm.first.joinToString(","))
        }
        if (nw == 0 || nb == 0) return LearnRes(false, "Не вижу и белых, и чёрных фигур — проверь рамку доски.")
        val aw = sw / nw
        val ab = sbk / nb
        if (aw <= ab + 8f) {
            return LearnRes(
                false,
                "Белые и чёрные фигуры не различились по яркости (белые ${aw.toInt()}, чёрные ${ab.toInt()}). " +
                        "Проверь, что «Играю за» совпадает с тем, кто стоит внизу доски."
            )
        }
        val newThr = (aw + ab) / 2f
        val ed = prefs.edit()
        for ((t, acc) in sums) {
            val n = cnt[t] ?: 1
            val avg = FloatArray(N * N) { acc[it] / n }
            tpl[t] = avg
            val by = ByteArray(N * N) { (avg[it] * 255f).toInt().coerceIn(0, 255).toByte() }
            ed.putString("tpl_$t", Base64.encodeToString(by, Base64.NO_WRAP))
        }
        thr = newThr
        ed.putFloat("bright_thr", newThr).putInt("tpl_ver", VER).commit()
        return LearnRes(true, "Эталоны сохранены автоматически ($samples фигур).")
    }

    // ---------- распознавание ----------

    private fun medCh(l: ArrayList<IntArray>, ch: Int): Int {
        val a = ArrayList<Int>()
        for (x in l) a.add(x[ch])
        return med(a)
    }

    /**
     * Скриншот доски -> FEN. Если на доске начальная позиция — эталоны фигур сохраняются сами
     * (первый раз, и потом обновляются при каждой новой партии, не чаще раза в 2 минуты).
     * На диск ничего не пишется, кроме самих эталонов в SharedPreferences.
     */
    fun recognize(bmp: Bitmap, playWhite: Boolean, prefs: SharedPreferences): Rec {
        if (ensureCv().not()) return Rec(null, "OpenCV не загрузился (initLocal вернул false)")
        val flip = playWhite.not()
        val cells = try {
            scanCells(bmp)
        } catch (e: Throwable) {
            return Rec(null, "Ошибка OpenCV: ${e.message}")
        }
        var occ = 0
        for (r in 0..7) for (c in 0..7) if (cells[r][c].occupied) occ++
        if (occ == 0) return Rec(null, "Фигур не найдено: проверь рамку доски («Доска»).")

        var note = ""
        val mm = startMismatch(cells, flip)
        val nearStart = mm.first.size + mm.second.size <= 4
        val strictStart = mm.first.isEmpty() && mm.second.isEmpty()
        val now = SystemClock.elapsedRealtime()
        if ((hasTemplates().not() && nearStart) || (strictStart && now - lastLearnAt > 120000L)) {
            val lr = learnFromCells(cells, flip, prefs)
            if (lr.ok) {
                note = lr.msg + " "
                lastLearnAt = now
            } else if (hasTemplates().not()) {
                return Rec(null, lr.msg)
            }
        }
        if (hasTemplates().not()) {
            return Rec(null, "Эталонов ещё нет: нажми «Сканировать» в начальной позиции — они сохранятся сами.")
        }

        val grid = CharArray(64) { '.' }
        val bad = ArrayList<String>()
        var count = 0
        var wk = 0
        var bk = 0
        for (r in 0..7) {
            for (c in 0..7) {
                val cell = cells[r][c]
                val m = cell.mask ?: continue
                val rr = if (flip) 7 - r else r
                val ff = if (flip) 7 - c else c
                var best = 0f
                var bt = 'p'
                for ((t, tp) in tpl) {
                    val s = softIou(tp, m)
                    if (s > best) {
                        best = s
                        bt = t
                    }
                }
                val name = sqName(rr, ff)
                if (best < MIN_SCORE) {
                    bad.add("$name (сходство ${(best * 100).toInt()}%)")
                    continue
                }
                if (bt == 'p' && (rr == 0 || rr == 7)) {
                    bad.add("$name (пешка на крайней горизонтали)")
                    continue
                }
                val piece = if (cell.bright > thr) bt.uppercaseChar() else bt
                if (piece == 'K') wk++
                if (piece == 'k') bk++
                grid[rr * 8 + ff] = piece
                count++
            }
        }
        if (bad.isNotEmpty()) return Rec(null, note + "Не распознаны клетки: " + bad.joinToString(", "))
        if (wk != 1) return Rec(null, note + (if (wk == 0) "Не найден белый король." else "Найдено белых королей: $wk."))
        if (bk != 1) return Rec(null, note + (if (bk == 0) "Не найден чёрный король." else "Найдено чёрных королей: $bk."))

        // кто ходил последним: подсвеченная клетка (последний ход) с фигурой
        val par0 = ArrayList<IntArray>()
        val par1 = ArrayList<IntArray>()
        for (r in 0..7) for (c in 0..7) {
            if ((r + c) % 2 == 0) par0.add(cells[r][c].bg) else par1.add(cells[r][c].bg)
        }
        val m0 = intArrayOf(medCh(par0, 0), medCh(par0, 1), medCh(par0, 2))
        val m1 = intArrayOf(medCh(par1, 0), medCh(par1, 1), medCh(par1, 2))
        var ourHi = 0
        var oppHi = 0
        for (r in 0..7) {
            for (c in 0..7) {
                val cell = cells[r][c]
                if (cell.mask == null) continue
                val ref = if ((r + c) % 2 == 0) m0 else m1
                var d = 0
                for (k in 0..2) d = maxOf(d, abs(cell.bg[k] - ref[k]))
                if (d > 16) {
                    val whitePiece = cell.bright > thr
                    if (whitePiece == playWhite) ourHi++ else oppHi++
                }
            }
        }
        val turn = if (oppHi > 0 && ourHi == 0) 1 else if (ourHi > 0 && oppHi == 0) 0 else -1

        var cs = ""
        if (grid[60] == 'K' && grid[63] == 'R') cs += "K"
        if (grid[60] == 'K' && grid[56] == 'R') cs += "Q"
        if (grid[4] == 'k' && grid[7] == 'r') cs += "k"
        if (grid[4] == 'k' && grid[0] == 'r') cs += "q"
        if (cs.isEmpty()) cs = "-"
        val fen = placement(grid) + (if (playWhite) " w " else " b ") + cs + " - 0 1"
        return Rec(fen, note + "Фигур: $count.", turn)
    }
}

/**
 * Доска по верхней и нижней границе (проведён один вертикальный свайп).
 * Если высота ≈ ширине экрана — доска на всю ширину (квадрат по центру свайпа).
 * Иначе — квадрат со стороной = высоте свайпа, по центру экрана.
 */
fun boardFromBand(y1: Float, y2: Float, screenW: Float): RectF {
    val top = minOf(y1, y2)
    val bottom = maxOf(y1, y2)
    val h = bottom - top
    return if (h >= screenW * 0.85f) {
        val t = ((top + bottom) / 2f - screenW / 2f).coerceAtLeast(0f)
        RectF(0f, t, screenW, t + screenW)
    } else {
        val left = (screenW - h) / 2f
        RectF(left, top, left + h, bottom)
    }
}

/** Рисует стрелку лучшего хода поверх доски. flip=true — доска перевёрнута (играем за чёрных). */
class ArrowView(ctx: Context) : View(ctx) {
    var board = RectF()
    var flip = false
    var move: Move? = null
    var showFrame = false

    private val line = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.argb(210, 0, 200, 83)
        style = Paint.Style.STROKE
        strokeCap = Paint.Cap.ROUND
    }
    private val fill = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.argb(210, 0, 200, 83)
        style = Paint.Style.FILL
    }
    private val frame = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.argb(200, 255, 193, 7)
        style = Paint.Style.STROKE
        strokeWidth = 3f
        pathEffect = DashPathEffect(floatArrayOf(18f, 12f), 0f)
    }

    private fun cx(sq: Int): Float {
        val f = sq % 8
        return board.left + ((if (flip) 7 - f else f) + 0.5f) * board.width() / 8f
    }

    private fun cy(sq: Int): Float {
        val r = sq / 8
        return board.top + ((if (flip) 7 - r else r) + 0.5f) * board.height() / 8f
    }

    override fun onDraw(c: Canvas) {
        if (board.width() < 8f || board.height() < 8f) return
        if (showFrame) c.drawRect(board, frame)
        val m = move ?: return
        val x1 = cx(m.from)
        val y1 = cy(m.from)
        val x2 = cx(m.to)
        val y2 = cy(m.to)
        val cell = board.width() / 8f
        line.strokeWidth = cell * 0.16f
        val ang = atan2((y2 - y1).toDouble(), (x2 - x1).toDouble())
        val head = cell * 0.4f
        val bx = x2 - (head * 0.8f * cos(ang)).toFloat()
        val by = y2 - (head * 0.8f * sin(ang)).toFloat()
        c.drawCircle(x1, y1, cell * 0.14f, fill)
        c.drawLine(x1, y1, bx, by, line)
        val p = Path()
        p.moveTo(x2, y2)
        val a1 = ang + Math.PI * 5 / 6
        val a2 = ang - Math.PI * 5 / 6
        p.lineTo(x2 + (head * cos(a1)).toFloat(), y2 + (head * sin(a1)).toFloat())
        p.lineTo(x2 + (head * cos(a2)).toFloat(), y2 + (head * sin(a2)).toFloat())
        p.close()
        c.drawPath(p, fill)
    }
}

/**
 * Полноэкранное затемнение. Нужно провести пальцем ТОЛЬКО сверху вниз:
 * от верхней до нижней границы доски. Ширина подгоняется под экран автоматически.
 * Область сохраняется при отпускании пальца. «Отмена» — внизу.
 */
class BoardSelectView(
    ctx: Context,
    private val onDone: (RectF) -> Unit,
    private val onCancel: () -> Unit
) : View(ctx) {
    private var y1 = 0f
    private var y2 = 0f
    private var drawing = false

    private val frame = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.argb(255, 0, 200, 83)
        style = Paint.Style.STROKE
        strokeWidth = 4f
    }
    private val inner = Paint().apply { color = Color.argb(50, 0, 200, 83) }
    private val bandLine = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.argb(255, 255, 193, 7)
        strokeWidth = 5f
    }
    private val txt = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.WHITE
        textSize = 42f
        textAlign = Paint.Align.CENTER
    }
    private val cancelBg = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = Color.argb(230, 200, 50, 50) }
    private val cancelRect = RectF()

    init {
        setBackgroundColor(Color.argb(150, 0, 0, 0))
    }

    override fun onDraw(c: Canvas) {
        c.drawText("Проведи пальцем сверху вниз:", width / 2f, 140f, txt)
        c.drawText("от верхнего до нижнего края доски", width / 2f, 200f, txt)
        cancelRect.set(width / 2f - 160f, height - 220f, width / 2f + 160f, height - 120f)
        c.drawRoundRect(cancelRect, 30f, 30f, cancelBg)
        c.drawText("Отмена", width / 2f, height - 155f, txt)
        if (drawing) {
            val r = boardFromBand(y1, y2, width.toFloat())
            c.drawRect(r, inner)
            c.drawRect(r, frame)
            c.drawLine(0f, y1, width.toFloat(), y1, bandLine)
            c.drawLine(0f, y2, width.toFloat(), y2, bandLine)
        }
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        when (e.action) {
            MotionEvent.ACTION_DOWN -> {
                if (cancelRect.contains(e.x, e.y)) {
                    onCancel()
                    return true
                }
                y1 = e.y
                y2 = e.y
                drawing = true
                invalidate()
            }
            MotionEvent.ACTION_MOVE -> {
                y2 = e.y
                invalidate()
            }
            MotionEvent.ACTION_UP -> {
                if (drawing.not()) return true
                y2 = e.y
                if (abs(y2 - y1) > 150f) onDone(boardFromBand(y1, y2, width.toFloat()))
                else {
                    drawing = false
                    invalidate()
                }
            }
        }
        return true
    }

/**
 * Служба специальных возможностей: делает ход на экране двумя нажатиями (откуда -> куда).
 * Включается вручную: Настройки -> Специальные возможности -> Chess AutoPlayer.
 */
class AutoPlayer : AccessibilityService() {
    companion object {
        @Volatile
        var instance: AutoPlayer? = null
    }

    private val handler = Handler(Looper.getMainLooper())

    override fun onServiceConnected() {
        super.onServiceConnected()
        instance = this
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {}

    override fun onInterrupt() {}

    override fun onUnbind(intent: Intent?): Boolean {
        instance = null
        return super.onUnbind(intent)
    }

    override fun onDestroy() {
        instance = null
        super.onDestroy()
    }

    private fun tap(x: Float, y: Float, cb: (Boolean) -> Unit) {
        val p = Path()
        p.moveTo(x, y)
        val g = GestureDescription.Builder()
            .addStroke(GestureDescription.StrokeDescription(p, 0L, 70L))
            .build()
        val ok = dispatchGesture(g, object : AccessibilityService.GestureResultCallback() {
            override fun onCompleted(gestureDescription: GestureDescription?) {
                cb(true)
            }

            override fun onCancelled(gestureDescription: GestureDescription?) {
                cb(false)
            }
        }, null)
        if (ok.not()) cb(false)
    }

    /** Нажать на клетку «откуда», через 250 мс на клетку «куда»; при превращении — ещё раз на «куда» (ферзь). */
    fun playMove(x1: Float, y1: Float, x2: Float, y2: Float, promo: Boolean, cb: (Boolean) -> Unit) {
        tap(x1, y1) { ok1 ->
            if (ok1.not()) {
                cb(false)
            } else {
                handler.postDelayed({
                    tap(x2, y2) { ok2 ->
                        if (ok2.not()) {
                            cb(false)
                        } else if (promo) {

package com.example.chessoverlay

import android.app.Activity
import android.content.Intent
import android.media.projection.MediaProjectionManager
import android.os.Build
import android.os.Bundle

/** Прозрачная activity: просит у системы разрешение на захват экрана и отдаёт результат в OverlayService. */
class CaptureActivity : Activity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        val mpm = getSystemService(MEDIA_PROJECTION_SERVICE) as MediaProjectionManager
        startActivityForResult(mpm.createScreenCaptureIntent(), 4242)
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == 4242 && resultCode == RESULT_OK && data != null) {
            val i = Intent(this, OverlayService::class.java)
                .setAction("chess.PROJECTION")
                .putExtra("code", resultCode)
                .putExtra("data", data)
            if (Build.VERSION.SDK_INT >= 26) startForegroundService(i) else startService(i)
        }
        finish()
    }
}
